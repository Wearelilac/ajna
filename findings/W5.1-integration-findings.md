### 2.1 Feature A: Wearable Resilience (Full Flow)

**Scenario:**
```typescript
describe('Feature A: Wearable Resilience', () => {
  it('complete flow: ingest → baseline → PRI → radar', async () => {
    // 1. Create subject with consent
    const subject = await createSubject({ age: 35, consentWearable: true });
    
    // 2. Ingest 30 days of wearable data
    const metrics = generateWearableData(30, 'stable');
    await POST('/v1/wearable/batch', { subjectId: subject.id, days: metrics });
    
    // 3. Run baseline worker (day 15)
    await runWorker('ajna.wearable.compute-baseline', { 
      subject_id: subject.id, 
      date: '2026-11-15' 
    });
    
    // 4. Verify baseline computed
    const baseline = await GET(`/v1/subjects/${subject.id}/baseline`);
    expect(baseline.metrics).toContain('hrv_rmssd_ms');
    expect(baseline.n_valid_days).toBeGreaterThanOrEqual(14);
    
    // 5. Verify PRI computed
    const pri = await GET(`/v1/subjects/${subject.id}/pri`);
    expect(pri.pri).toBeGreaterThan(0);
    expect(pri.pri).toBeLessThanOrEqual(100);
    
    // 6. Verify radar data (mobile API)
    const radar = await GET(`/v1/subjects/${subject.id}/radar`);
    expect(radar.pri).toBeDefined();
    expect(radar.flags).toBeArrayOfSize(0); // Stable profile, no flags
    expect(radar.metrics).toBeDefined();
  });
});
```

### 2.2 Feature B: Acoustic Monitoring (Full Flow)

**Scenario:**
```typescript
describe('Feature B: Acoustic Monitoring', () => {
  it('complete flow: session → flag → review → outcome', async () => {
    // 1. Create subject with monitor consent
    const subject = await createSubject({ age: 25, consentMonitor: true });
    
    // 2. Start monitor session (mobile)
    const session = await POST('/v1/monitor/sessions', { subjectId: subject.id });
    expect(session.id).toBeDefined();
    expect(session.status).toBe('active');
    
    // 3. Send flags (7-field allow-list only)
    const flags = [
      {
        signature: 'S3',
        confidence: 0.75,
        timeBucket: '2026-11-14T14:00:00Z',
        modelVersion: 'acoustic-v1.0',
        sessionDurationSec: 120,
        energyDb: -35,
      },
      {
        signature: 'S3',
        confidence: 0.82,
        timeBucket: '2026-11-14T14:02:00Z',
        modelVersion: 'acoustic-v1.0',
        sessionDurationSec: 240,
        energyDb: -32,
      },
    ];
    
    await POST('/v1/monitor/flags', { sessionId: session.id, flags });
    
    // 4. Stop session
    await POST(`/v1/monitor/sessions/${session.id}/stop`);
    
    // 5. Verify deidentified flags stored
    const storedFlags = await GET('/v1/monitor/flags', { sessionId: session.id });
    expect(storedFlags.length).toBe(2);
    expect(storedFlags[0].signature).toBe('S3');
    
    // 6. Create crisis review (worker)
    await runWorker('ajna.crisis-review.create', { session_id: session.id });
    
    // 7. Assign to reviewer
    const reviews = await GET('/v1/crisis-reviews', { state: 'pending' });
    const review = reviews[0];
    
    await POST(`/v1/crisis-reviews/${review.id}/claim`, { reviewerId: 'reviewer-1' });
    
    // 8. Transition to in_review
    await POST(`/v1/crisis-reviews/${review.id}/transition`, {
      toState: 'in_review',
      reviewerId: 'reviewer-1',
    });
    
    // 9. Complete review with outcome
    await POST(`/v1/crisis-reviews/${review.id}/transition`, {
      toState: 'completed',
      reviewerId: 'reviewer-1',
      outcome: 'contact_scheduled',
      notes: 'Follow-up scheduled for next week',
    });
    
    // 10. Verify audit trail
    const events = await GET(`/v1/crisis-reviews/${review.id}/events`);
    expect(events.length).toBeGreaterThanOrEqual(3); // claim, in_review, completed
  });
});
```

### 2.3 Feature C: Population Health (Full Flow)

**Scenario:**
```typescript
describe('Feature C: Population Health', () => {
  it('complete flow: aggregate → detect gap → match → draft → review', async () => {
    // 1. Create 15 subjects in community (K≥10)
    const subjects = await createSubjects(15, { 
      community: 'COMM_TEST',
      age_band: '19-50',
      consentPopulation: true 
    });
    
    // 2. Ingest wearable data for all subjects
    for (const subject of subjects) {
      await ingestWearableData(subject.id, 30);
    }
    
    // 3. Run population aggregation
    await runWorker('ajna.population.aggregate', { 
      granularity: 'week',
      week_start: '2026-11-11' // Monday
    });
    
    // 4. Verify aggregates (K≥10, not suppressed)
    const agg = await GET('/v1/population/communities', { communityId: 'COMM_TEST' });
    expect(agg[0].suppressed).toBe(false);
    expect(agg[0].subjectsN).toBeGreaterThanOrEqual(10);
    
    // 5. Run care-gap detection (overdue screenings)
    await runWorker('ajna.caregap.detect', { gap_types: ['overdue_screening'] });
    
    // 6. Verify gaps detected
    const gaps = await GET('/v1/care-gaps', { status: 'detected' });
    expect(gaps.length).toBeGreaterThan(0);
    
    // 7. Run pathway matching
    await runWorker('ajna.caregap.match', { care_gap_id: gaps[0].id, top_k: 3 });
    
    // 8. Verify matches
    const matches = await GET(`/v1/care-gaps/${gaps[0].id}/matches`);
    expect(matches.length).toBeGreaterThan(0);
    expect(matches[0].similarity).toBeGreaterThan(0.5);
    
    // 9. Run draft generation
    await runWorker('ajna.caregap.draft', { 
      care_gap_id: gaps[0].id,
      pathway_id: matches[0].pathwayId
    });
    
    // 10. Verify draft generated
    const drafts = await GET('/v1/outreach-drafts', { status: 'draft' });
    expect(drafts.length).toBeGreaterThan(0);
    expect(drafts[0].body).toContain('screening'); // Template wording
    
    // 11. Review workflow: draft → in_review → approved
    const draft = drafts[0];
    
    await POST(`/v1/outreach-drafts/${draft.id}/transition`, {
      toStatus: 'in_review',
    });
    
    await POST(`/v1/outreach-drafts/${draft.id}/transition`, {
      toStatus: 'approved',
      reason: 'Approved for delivery',
    });
    
    // 12. Verify final state
    const approvedDraft = await GET(`/v1/outreach-drafts/${draft.id}`);
    expect(approvedDraft.status).toBe('approved');
    
    // 13. Verify NO automatic send (audit)
    const networkLogs = await getNetworkLogs();
    expect(networkLogs).not.toContain('twilio');
    expect(networkLogs).not.toContain('sendgrid');
  });
});
```

### 3.2 Consent Interplay Test

```typescript
describe('Consent Revocation (All Features)', () => {
  it('revoke wearable_use → stops baseline computation', async () => {
    const subject = await createSubject({ consentWearable: true });
    
    // Ingest data
    await ingestWearableData(subject.id, 30);
    
    // Revoke consent
    await PUT(`/v1/subjects/${subject.id}/consent`, { wearable_use: false });
    
    // Attempt baseline computation
    await expect(runWorker('ajna.wearable.compute-baseline', { 
      subject_id: subject.id, 
      date: '2026-11-15' 
    })).rejects.toThrow('Consent revoked');
  });
  
  it('revoke population_use → excludes from aggregates', async () => {
    const subject = await createSubject({ consentPopulation: true, community: 'COMM_TEST' });
    
    // Ingest data
    await ingestWearableData(subject.id, 30);
    
    // Run aggregation (subject included)
    await runWorker('ajna.population.aggregate', { week_start: '2026-11-11' });
    const agg1 = await GET('/v1/population/communities/COMM_TEST');
    const count1 = agg1.subjectsN;
    
    // Revoke consent
    await PUT(`/v1/subjects/${subject.id}/consent`, { population_use: false });
    
    // Re-run aggregation (subject excluded)
    await runWorker('ajna.population.aggregate', { week_start: '2026-11-18' });
    const agg2 = await GET('/v1/population/communities/COMM_TEST');
    const count2 = agg2.subjectsN;
    
    expect(count2).toBe(count1 - 1);
  });
});
```
