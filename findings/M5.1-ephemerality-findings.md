### 4.2 Battery Optimization

**Inference Throttling:**
```typescript
// apps/mobile/src/edge/InferenceEngine.ts

class InferenceEngine {
  private batteryLevel: number = 1.0;
  
  async infer(samples: Float32Array): Promise<InferenceResult> {
    // Throttle inference based on battery level
    if (this.batteryLevel < 0.2) {
      // Low battery: infer every 5 seconds
      if (Date.now() - this.lastInferenceTime < 5000) {
        return null;
      }
    } else if (this.batteryLevel < 0.5) {
      // Medium battery: infer every 2 seconds
      if (Date.now() - this.lastInferenceTime < 2000) {
        return null;
      }
    }
    
    // Normal: infer every window (default)
    return this.runInference(samples);
  }
}
```

### 4.3 Error Handling

**Network Failures:**
```typescript
// apps/mobile/src/api/monitor.ts

async function sendFlag(flag: Flag): Promise<void> {
  try {
    await fetch(`${API_BASE}/v1/monitor/flags`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(flag),
    });
  } catch (error) {
    // Queue for retry (store in AsyncStorage)
    await queueFailedFlag(flag);
  }
}

async function retryFailedFlags(): Promise<void> {
  const queuedFlags = await getQueuedFlags();
  
  for (const flag of queuedFlags) {
    try {
      await sendFlag(flag);
      await removeQueuedFlag(flag.id);
    } catch (error) {
      // Keep in queue for next retry
    }
  }
}
```

## 6. Known Limitations

### 6.1 No Background Audio on iOS (Backgrounded)
- iOS suspends audio when app is backgrounded (unless specific entitlements)
- Mitigation: Document in user guide, recommend foreground use

### 6.2 No Wake-Lock on Android
- Screen may turn off during session, pausing audio
- Mitigation: Add wake-lock permission (WAKE_LOCK) for active sessions
